cmake_minimum_required(VERSION 3.10)
project(arp_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCAP REQUIRED libpcap)
add_executable(arp_lab src/main.cpp src/arp_monitor.cpp src/arp_simulator.cpp)
target_include_directories(arp_lab PRIVATE ${PCAP_INCLUDE_DIRS} include)
target_link_libraries(arp_lab PRIVATE ${PCAP_LIBRARIES})
